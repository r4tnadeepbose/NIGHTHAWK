#!/usr/bin/env bash
# NIGHTHAWK Linux prerequisite checker and installer (not a pip requirements file).

set -u

green='\033[0;32m'
red='\033[0;31m'
yellow='\033[0;33m'
reset='\033[0m'
missing=0

ok()   { printf "${green}[OK]${reset} %s\n" "$1"; }
miss() { printf "${red}[MISS]${reset} %s\n" "$1"; missing=1; }
info() { printf "${yellow}[..]${reset} %s\n" "$1"; }

info "Checking NIGHTHAWK requirements"

if command -v python3 >/dev/null 2>&1; then
    python_version="$(python3 -c 'import sys; print("%d.%d" % sys.version_info[:2])')"
    if python3 -c 'import sys; raise SystemExit(sys.version_info < (3, 9))'; then
        ok "python3 $python_version"
    else
        miss "python3 $python_version (Python 3.9 or newer is required)"
    fi
else
    miss "python3 not found"
fi

if command -v python3 >/dev/null 2>&1 && python3 -c 'import tkinter' >/dev/null 2>&1; then
    ok "tkinter is importable"
else
    miss "tkinter is not importable"
fi

if command -v nmap >/dev/null 2>&1; then
    nmap_version="$(nmap --version 2>/dev/null | head -n 1)"
    ok "${nmap_version:-nmap found}"
else
    miss "nmap not found on PATH"
fi

if [ "$missing" -eq 0 ]; then
    ok "All NIGHTHAWK requirements are available"
    exit 0
fi

if [ "$(id -u)" -ne 0 ]; then
    info "Missing packages require administrator privileges. Run: sudo ./requirements.txt"
    exit 1
fi

info "Installing missing packages"
if command -v apt >/dev/null 2>&1; then
    apt update
    apt install -y nmap python3 python3-tk
elif command -v dnf >/dev/null 2>&1; then
    dnf install -y nmap python3 python3-tkinter
elif command -v brew >/dev/null 2>&1; then
    brew install nmap python3 python-tk
else
    miss "No supported package manager found (apt, dnf, or brew)"
    exit 1
fi

info "Rechecking installed requirements"
exec "$0"
